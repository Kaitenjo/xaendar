my-app/
├── package.json        scripts: start (xd start), build (xd build), test (xd test)
├── xaendar.json        project settings read by the CLI
├── tsconfig.json       also used to type-check the templates
├── vite.config.ts      plain Vite config: the CLI adds the Xaendar plugins
└── src/
    ├── index.html
    ├── main.ts         installs the Signal polyfill, then loads bootstrap.ts
    ├── bootstrap.ts    imports the components, so that they get defined
    ├── styles.css      global styles (html, body); .scss, .sass or .less with another format
    └── my-app-root/
        ├── my-app-root.xd.component.ts
        ├── my-app-root.xd.component.html
        ├── my-app-root.xd.component.css
        └── my-app-root.xd.component.spec.ts
